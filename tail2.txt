const ALPHABET: &[u8] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

fn hash_addr5d(addr5d: &[u8]) -> [u8; 16] {
    use blake2::Blake2s;
    use blake2::digest::{Update, VariableOutput};

    let mut hasher = Blake2s::new(16).expect("Blake2s init");
    hasher.update(addr5d);
    let mut out = [0u8; 16];
    hasher.finalize_variable(|res| out.copy_from_slice(res));
    out
}

fn encode_shortcode_from_bytes(addr5d: &[u8]) -> Vec<u8> {
    let digest = hash_addr5d(addr5d);

    let mut value = 0u128;
    for b in digest.iter() {
        value = (value << 8) | (*b as u128);
    }

    let mut chars = Vec::new();
    for _ in 0..5 {
        let idx = (value & 0x1F) as usize;
        chars.push(ALPHABET[idx]);
        value >>= 5;
    }

    chars.reverse();
    chars
}

pub struct VaultShortcodeMigration<T>(sp_std::marker::PhantomData<T>);

impl<T: Config> frame_support::traits::OnRuntimeUpgrade for VaultShortcodeMigration<T> {
    fn on_runtime_upgrade() -> frame_support::weights::Weight {
        use frame_support::storage::StorageMap;
        use frame_support::traits::Get;

        let mut weight: frame_support::weights::Weight = 0;

        VaultSecrets::<T>::translate(|_id, mut desc: VaultSecretDescriptor<T>| {
            let shortcode_vec = encode_shortcode_from_bytes(&desc.addr5d);

            if let Ok(bounded) =
                BoundedVec::<u8, T::MaxShortcodeLength>::try_from(shortcode_vec)
            {
                desc.shortcode = bounded;
            }

            weight += T::DbWeight::get().reads_writes(1, 1);
            Some(desc)
        });

        weight
    }
}
