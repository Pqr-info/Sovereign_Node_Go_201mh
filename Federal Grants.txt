🔄 3. Update Loop (Core REPL Logic)
This is the heart of the REPL.
It handles user input, dispatches to the engine, and updates state.

go
func (m Model) Update(msg bubbletea.Msg) (bubbletea.Model, bubbletea.Cmd) {

    switch msg := msg.() {

    case InputMsg:
        // Compile prompt
        input := msg.Text
        m.input = ""

        compiled := prompt.Compile(input, m.ticketSet, m.agentMode)

        // Dispatch to engine
        return m, func() bubbletea.Msg {
            out, err := m.engine.Invoke(m.ctx, compiled)
            if err != nil {
                return ErrorMsg{Err: err}
            }
            return EngineResponseMsg{Output: out}
        }

    case EngineResponseMsg:
        m.lastOutput = msg.Output
        return m, nil

    case TicketUpdatedMsg:
        // Tickets changed → refresh view
        return m, nil

    case AgentModeChangedMsg:
        // Mode changed → refresh view
        return m, nil

    case MemoryEventMsg:
        // Memory graph emitted an event
        m.lastOutput = renderMemoryEvent(msg)
        return m, nil

    case ErrorMsg:
        m.lastOutput = "Error: " + msg.Err.Error()
        return m, nil

    default:
        return m, nil
    }
}

    var sb strings.Builder

func renderConsensusAnalysis(c chat.ConsensusOutput) string {
    // Simple heuristic: count shared phrases
    agreement := 0
    divergence := 0

    agents := make([]string, 0, len(c.AgentOutputs))
    for a := range c.AgentOutputs {
        agents = append(agents, a)
    }

    for i := 0; i < len(agents); i++ {
        for j := i + 1; j < len(agents); j++ {
            a := c.AgentOutputs[agents[i]]
            b := c.AgentOutputs[agents[j]]

            if strings.Contains(a, "race condition") && strings.Contains(b, "race condition") {
                agreement++
            } else {
                divergence++
            }
        }
    }

    return fmt.Sprintf(
        "- Agreement Score: %d\n- Divergence Score: %d\n\n",
        agreement,
        divergence,
    )
}
